# CONTENT SIGNALS (https://contentsignals.org/) — tuyên bố cách nội dung được dùng:
#   search=yes   : được lập chỉ mục và hiện trong kết quả tìm kiếm.
#   ai-input=yes : được đưa vào AI để trả lời người dùng (RAG, AI Overviews),
#                  kèm dẫn nguồn.
#   ai-train=no  : KHÔNG đồng ý dùng để huấn luyện hay tinh chỉnh mô hình.
# Đây là tín hiệu thể hiện mong muốn, không phải cơ chế chặn: bot vẫn crawl
# được như các nhóm `Allow: /` bên dưới.
#
# Dòng Content-Signal phải lặp lại trong TỪNG nhóm: mỗi bot chỉ đọc nhóm
# User-agent cụ thể nhất khớp với nó, nên GPTBot sẽ không thấy tín hiệu đặt ở
# nhóm `*`.

# ---------------------------------------------------------------------------
# Mặc định: mọi bot đều được crawl toàn site.
# ---------------------------------------------------------------------------
User-agent: *
Content-Signal: search=yes, ai-input=yes, ai-train=no
Allow: /

# ---------------------------------------------------------------------------
# AI / LLM crawler — khai báo TƯỜNG MINH thay vì để ngầm theo `User-agent: *`.

# robots.txt; liệt kê ra để không bị bỏ sót, và để ý định "cho phép" được ghi
# rõ ràng thay vì phải suy luận.
#
# Không có nhóm nào ở đây thu hẹp quyền so với `User-agent: *` bên trên.
# ---------------------------------------------------------------------------
